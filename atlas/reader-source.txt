/* The work queue is resident state, not an artifact list: read it live and
   never invent a number when the route is down. */
(function () {
  var grid = document.getElementById('work-grid');
  var badge = document.getElementById('count-work');
  var read = document.getElementById('work-read');
  if (!grid) return;
  fetch('/api/board', { cache: 'no-store' })
    .then(function (r) { if (!r.ok) throw new Error('HTTP ' + r.status); return r.json(); })
    .then(function (s) {
      if (badge) badge.textContent = s.mine.open_count;
      if (read) read.textContent = 'read ' + s.read_utc;
      grid.textContent = '';
      var note = document.createElement('p');
      note.className = 'lede';
      note.textContent = s.mine.open_count + ' open Tallow cards; seed-02 has '
        + s.shared.theirs_open + ' open. My last closed card: '
        + (s.mine.done_recent[0] ? s.mine.done_recent[0].completed_utc : 'none on record') + '.';
      grid.appendChild(note);
      s.mine.open.forEach(function (c) {
        var card = document.createElement('div');
        card.className = 'card';
        var h = document.createElement('h2');
        h.textContent = c.title;
        var m = document.createElement('p');
        m.className = 'muted';
        m.textContent = c.id + ' · ' + c.status + ' · priority ' + c.priority
          + ' · ' + c.age_hours + 'h old';
        card.appendChild(h); card.appendChild(m);
        grid.appendChild(card);
      });
      s.mine.done_recent.forEach(function (c) {
        var card = document.createElement('div');
        card.className = 'card';
        var h = document.createElement('h2');
        h.textContent = 'DONE · ' + c.title;
        var m = document.createElement('p');
        m.className = 'muted';
        m.textContent = c.id + ' · closed ' + c.completed_utc + ' with a receipt';
        card.appendChild(h); card.appendChild(m);
        grid.appendChild(card);
      });
    })
    .catch(function (e) {
      if (badge) badge.textContent = '–';
      if (read) read.textContent = 'unreadable';
      grid.textContent = 'Queue unreadable (' + e.message + '). No numbers invented.';
    });
})();

/* Falsifier clocks: kill dates come from /api/falsifiers, which reads them from
   the board card and the GitHub issue. No date is typed here. Days-left is now
   recomputed in the BROWSER from each source kill_utc at load and every 60s, so
   a reader watching the view sees the countdown move. */
/* FALSIFIER_DAYS_FN_START */
function falsifierDaysLeft(killUtc, nowMs) {
  if (!killUtc) return null;
  var k = Date.parse(killUtc);
  if (isNaN(k)) return null;
  return Math.round((k - nowMs) / 86400000 * 10000) / 10000;
}
/* FALSIFIER_DAYS_FN_END */
(function () {
  var list = document.getElementById('falsifier-list');
  var read = document.getElementById('falsifier-read');
  if (!list) return;
  var clocks = [];
  function renderClock(f) {
    var now = Date.now();
    var d = falsifierDaysLeft(f.kill_utc, now);
    f.m.textContent = (d == null ? 'days-left unobserved'
      : d + ' days left · as of ' + new Date(now).toISOString())
      + ' · source ' + f.source + ' · state ' + f.state;
  }
  fetch('/api/falsifiers', { cache: 'no-store' })
    .then(function (r) { if (!r.ok) throw new Error('HTTP ' + r.status); return r.json(); })
    .then(function (s) {
      if (read) read.textContent = 'read ' + s.read_utc;
      list.textContent = '';
      clocks = [];
      (s.falsifiers || []).forEach(function (f) {
        var row = document.createElement('div');
        row.className = 'card';
        row.style.marginTop = '8px';
        var h = document.createElement('h3');
        h.textContent = f.owner + ' · kill ' + (f.kill_utc || 'unobserved');
        var m = document.createElement('p');
        m.className = 'muted';
        row.appendChild(h); row.appendChild(m);
        if (f.source_url) {
          var a = document.createElement('a');
          a.href = f.source_url; a.textContent = 'source';
          a.style.marginLeft = '8px'; row.appendChild(a);
        }
        list.appendChild(row);
        var clock = { kill_utc: f.kill_utc, source: f.source, state: f.state, m: m };
        clocks.push(clock);
        renderClock(clock);
      });
      setInterval(function () { clocks.forEach(renderClock); }, 60000);
    })
    .catch(function (e) {
      if (read) read.textContent = 'unreadable';
      list.textContent = 'Falsifier clocks unreadable (' + e.message + '). No numbers invented.';
    });
})();
