/* The work queue is resident state, not an artifact list: read it live and
   never invent a number when the route is down. */
(function () {
  var grid = document.getElementById('work-grid');
  var summary = document.getElementById('work-summary');
  var badge = document.getElementById('count-work');
  var read = document.getElementById('work-read');
  if (!grid) return;
  fetch('/api/board', { cache: 'no-store' })
    .then(function (r) { if (!r.ok) throw new Error('HTTP ' + r.status); return r.json(); })
    .then(function (s) {
      if (read) read.textContent = 'read ' + s.read_utc;
      grid.textContent = '';
      if (summary) summary.textContent = '';
      if (!s.owners) {
        if (summary) summary.textContent = 'Board snapshot is outdated (schema '
          + (s.schema || 'unknown') + '); no cards shown.';
        return;
      }
      var names = { 'default': 'Tallow', 'seed-02': 'Cairn (seed-02)', unassigned: 'Unassigned' };
      var labels = { ready: 'READY', in_progress: 'IN PROGRESS',
                     running: 'RUNNING', blocked: 'BLOCKED',
                     scheduled: 'SCHEDULED (future-dated, not open work)' };
      // A group owns its own grid. Putting an <h3> or the summary directly in
      // the outer grid makes them cells beside cards — the ragged layout Niko
      // saw in the 27 Nov screenshot.
      function group(title, cards, isDone) {
        var section = document.createElement('section');
        section.className = 'work-group';
        var h = document.createElement('h3');
        h.className = 'section-label';
        h.textContent = title;
        section.appendChild(h);
        var inner = document.createElement('div');
        inner.className = 'card-grid';
        cards.forEach(function (c) {
          var el = document.createElement('div');
          el.className = 'card';
          var t = document.createElement('h2');
          t.textContent = c.title;
          var m = document.createElement('p');
          m.className = 'muted';
          m.textContent = isDone
            ? c.id + ' · closed ' + c.completed_utc + ' with a receipt'
            : c.id + ' · priority ' + c.priority + ' · ' + c.age_hours + 'h old'
              + (c.status === 'blocked' && c.block_kind ? ' · ' + c.block_kind : '');
          el.appendChild(t); el.appendChild(m);
          inner.appendChild(el);
        });
        section.appendChild(inner);
        grid.appendChild(section);
      }
      if (badge) badge.textContent = s.totals.open;
      if (summary) summary.textContent = s.totals.open + ' open · '
        + (s.totals.scheduled || 0) + ' scheduled · Tallow '
        + s.owners['default'].open_count + ' · Cairn '
        + s.owners['seed-02'].open_count + ' · unassigned '
        + s.owners.unassigned.open_count + '. Read ' + s.read_utc
        + '. Scheduled cards are future-dated commitments, counted apart from open work. '
        + 'Open cards first; closes show the last five per owner. '
        + 'A card closes only with a receipt.';
      Object.keys(names).forEach(function (owner) {
        var o = s.owners[owner];
        if (!o.open_count && !o.done_count && !o.scheduled_count) return;
        s.states.forEach(function (st) {
          var cards = o.states[st];
          if (cards.length) {
            group(names[owner] + ' · ' + labels[st] + ' (' + cards.length + ')', cards, false);
          }
        });
        if (o.done_recent.length) {
          group(names[owner] + ' · DONE · last ' + o.done_recent.length, o.done_recent, true);
        }
      });
    })
    .catch(function (e) {
      if (badge) badge.textContent = '–';
      if (read) read.textContent = 'unreadable';
      grid.textContent = 'Queue unreadable (' + e.message + '). No numbers invented.';
    });
})();

/* Falsifier clocks: kill dates come from /api/falsifiers, which reads them from
   the board card and the GitHub issue. No date is typed here. The snapshot is
   RE-FETCHED on the same 60s cadence as the countdown, so a kill date, owner,
   source or read-date that changes at source propagates to the reader rather
   than staying frozen at page load. Nothing here closes, extends or invents a
   kill date, and nothing spends. */
/* FALSIFIER_DAYS_FN_START */
function falsifierDaysLeft(killUtc, nowMs) {
  if (!killUtc) return null;
  var k = Date.parse(killUtc);
  if (isNaN(k)) return null;
  return Math.round((k - nowMs) / 86400000 * 10000) / 10000;
}
/* FALSIFIER_DAYS_FN_END */
/* FALSIFIER_LABEL_FN_START */
function falsifierLabel(killUtc, daysLeft, nowMs) {
  // A kill date already past is ELAPSED, not a negative countdown: name the
  // source date and say so. A live one counts down. Unobserved stays unobserved.
  // No number here is typed by hand: every value comes from the snapshot/clock.
  if (!killUtc || daysLeft == null) return 'days-left unobserved';
  if (daysLeft < 0) return 'elapsed · kill date ' + killUtc + ' · as of ' + new Date(nowMs).toISOString();
  return daysLeft + ' days left · as of ' + new Date(nowMs).toISOString();
}

  function renderClock(f) {
    var now = Date.now();
    var d = falsifierDaysLeft(f.kill_utc, now);
    f.h.textContent = f.owner + ' · kill ' + (f.kill_utc || 'unobserved');
    f.m.textContent = falsifierLabel(f.kill_utc, d, now)
      + ' · source ' + f.source + ' · state ' + f.state;
  }
  function loadFalsifiers() {
    return fetch('/api/falsifiers', { cache: 'no-store' })
      .then(function (r) { if (!r.ok) throw new Error('HTTP ' + r.status); return r.json(); })
      .then(function (s) {
        lastRead = s.read_utc;
        if (read) read.textContent = 'read ' + lastRead;
        list.textContent = '';
        clocks = [];
        (s.falsifiers || []).forEach(function (f) {
          var row = document.createElement('div');
          row.className = 'card';
          row.style.marginTop = '8px';
          var h = document.createElement('h3');
          var m = document.createElement('p');
          m.className = 'muted';
          row.appendChild(h); row.appendChild(m);
          if (f.source_url) {
            var a = document.createElement('a');
            a.href = f.source_url; a.textContent = 'source';
            a.style.marginLeft = '8px'; row.appendChild(a);
          }
          list.appendChild(row);
          var clock = { owner: f.owner, kill_utc: f.kill_utc, source: f.source,
                        state: f.state, h: h, m: m };
          clocks.push(clock);
          renderClock(clock);
        });
      });
  }
  loadFalsifiers().catch(function (e) {
    if (read) read.textContent = 'unreadable';
    list.textContent = 'Falsifier clocks unreadable (' + e.message + '). No numbers invented.';
  });
  // ONE timer: refresh the source snapshot AND recompute days-left every 60s.
  // A failed refresh keeps the last rendered rows; no number is invented.
  setInterval(function () {
    loadFalsifiers().catch(function (e) {
      if (read) read.textContent = 'refresh unreadable (last read ' + (lastRead || 'unknown') + ')';
    });
    clocks.forEach(renderClock);
  }, 60000);
})();
